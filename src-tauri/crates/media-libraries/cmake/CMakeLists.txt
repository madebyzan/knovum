cmake_minimum_required(VERSION 4.3)
project(KnovumMediaLibraries LANGUAGES C CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source.json" source)
string(JSON supplier GET "${source}" targets ${KNOVUM_TARGET})
function(download record output)
  string(JSON url GET "${record}" url)
  string(JSON digest GET "${record}" sha256)
  file(DOWNLOAD "${url}" "${output}" EXPECTED_HASH "SHA256=${digest}" TLS_VERIFY ON STATUS result)
  list(GET result 0 code)
  if(NOT code EQUAL 0)
    message(FATAL_ERROR "Media supplier download failed: ${result}")
  endif()
endfunction()
download("${supplier}" "${CMAKE_BINARY_DIR}/ffmpeg.zip")
file(ARCHIVE_EXTRACT INPUT "${CMAKE_BINARY_DIR}/ffmpeg.zip" DESTINATION "${CMAKE_BINARY_DIR}/supplier")
string(JSON entry GET "${supplier}" entry)
if(WIN32)
  set(name ffmpeg.exe)
else()
  set(name ffmpeg)
endif()
file(MAKE_DIRECTORY "${CMAKE_INSTALL_PREFIX}/bin" "${CMAKE_INSTALL_PREFIX}/notices" "${CMAKE_INSTALL_PREFIX}/build-evidence")
set(executable "${CMAKE_INSTALL_PREFIX}/bin/${name}")
file(COPY_FILE "${CMAKE_BINARY_DIR}/supplier/${entry}" "${executable}")
file(CHMOD "${executable}" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
string(JSON license GET "${source}" licenseSource)
download("${license}" "${CMAKE_INSTALL_PREFIX}/notices/COPYING.GPLv3")
string(JSON count ERROR_VARIABLE absent LENGTH "${supplier}" noticeSources)
if(NOT absent)
  math(EXPR last "${count}-1")
  foreach(i RANGE ${last})
    string(JSON notice GET "${supplier}" noticeSources ${i})
    string(JSON name GET "${notice}" fileName)
    download("${notice}" "${CMAKE_INSTALL_PREFIX}/notices/${name}")
  endforeach()
endif()
string(JSON count ERROR_VARIABLE absent LENGTH "${supplier}" archiveNotices)
if(NOT absent)
  math(EXPR last "${count}-1")
  foreach(i RANGE ${last})
    string(JSON name MEMBER "${supplier}" archiveNotices ${i})
    string(JSON entry GET "${supplier}" archiveNotices ${name})
    file(COPY_FILE "${CMAKE_BINARY_DIR}/supplier/${entry}" "${CMAKE_INSTALL_PREFIX}/notices/${name}")
  endforeach()
endif()
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/source.json" DESTINATION "${CMAKE_INSTALL_PREFIX}/build-evidence")
