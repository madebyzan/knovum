cmake_minimum_required(VERSION 3.20)
project(KnovumVoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_MODEL_SET custom CACHE STRING "" FORCE)
set(AUDIOCPP_DEPLOYMENT_BUILD ON CACHE BOOL "" FORCE)
set(AUDIOCPP_MODELS "voxcpm2;qwen3_forced_aligner" CACHE STRING "" FORCE)
set(AUDIOCPP_BUILD_C_API ON CACHE BOOL "" FORCE)
set(AUDIOCPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(AUDIOCPP_STATIC_ESPEAK OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_NATIVE_CPU OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
    set(ENGINE_ENABLE_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
endif()
set(ENGINE_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(APPLE)
    set(GGML_METAL_EMBED_LIBRARY OFF CACHE BOOL "" FORCE)
endif()
# The supplier tree builds under "s" so its own nested ExternalProject paths stay inside the
# 260-character limit that MSVC applies to intermediate build files.
add_subdirectory("${KNOVUM_AUDIOCPP_SOURCE}" s EXCLUDE_FROM_ALL)
if(GGML_BACKEND_DL)
    set(KNOVUM_BACKEND_TARGETS ggml ggml-base ${GGML_AVAILABLE_BACKENDS})
    if(APPLE)
        set_target_properties(${KNOVUM_BACKEND_TARGETS} PROPERTIES INSTALL_RPATH "@loader_path")
    endif()
    install(TARGETS ${KNOVUM_BACKEND_TARGETS} RUNTIME DESTINATION bin COMPONENT knovum-voice LIBRARY DESTINATION bin COMPONENT knovum-voice)
endif()
set_target_properties(audiocpp PROPERTIES PREFIX "" VERSION "" SOVERSION "")
if(APPLE AND GGML_BACKEND_DL)
    set_target_properties(audiocpp PROPERTIES INSTALL_RPATH "@loader_path/../bin")
endif()
install(TARGETS audiocpp RUNTIME DESTINATION compiled-libraries COMPONENT knovum-voice LIBRARY DESTINATION compiled-libraries COMPONENT knovum-voice)
add_library(knovum_voice SHARED "${KNOVUM_VOICE_BRIDGE}")
target_include_directories(knovum_voice PRIVATE "${KNOVUM_AUDIOCPP_SOURCE}/include")
target_link_libraries(knovum_voice PRIVATE engine_runtime ggml)
set_target_properties(knovum_voice PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
if(APPLE AND GGML_BACKEND_DL)
    set_target_properties(knovum_voice PROPERTIES INSTALL_RPATH "@loader_path/../bin")
endif()
install(TARGETS knovum_voice RUNTIME DESTINATION compiled-libraries COMPONENT knovum-voice LIBRARY DESTINATION compiled-libraries COMPONENT knovum-voice)
if(APPLE)
    add_dependencies(knovum_voice ggml-metal-lib)
    install(FILES "${CMAKE_BINARY_DIR}/bin/default.metallib" DESTINATION bin COMPONENT knovum-voice)
endif()
